# Signed distance function for a star

First we need a way to render an sdf in a P5 sketch

# %%--- [javascript]
# properties:
#   top_hidden: true
#   run_on_load: true
# ---%%
var p5WithCleanup = (sketch, text) => {
    if (!window.sketches) {
        window.sketches = {};
    }

    if (window.sketches[text]) {
        window.sketches[text].remove();
    }

    const matchedElements = Array.from(
        document.querySelectorAll('.cell-bottom'),
    );

    const matchedElement = matchedElements.find(
        (elm) => elm.parentElement.textContent.split(text).length > 1,
    );

    if (!matchedElement) {
        throw new Error(`p5WithCleanup: text not found '${text}'`);
    }

    const p5Instance = new p5(sketch, matchedElement);
    window.sketches[text] = p5Instance;
    return p5Instance;
};

# %%--- [javascript]
# properties:
#   run_on_load: true
# ---%%
await import('/p5-notebook/lib/p5/lib/p5.js');

var notImplemented = () => {
    throw new Error('Not implemented');
};

var createVector = notImplemented;
var map = notImplemented;
var constrain = notImplemented;

var createSetupFunction = (p, sdf) => () => {
    createVector = p.createVector;
    map = p.map;
    constrain = p.constrain;

    p.createCanvas(300, 300);
    p.colorMode(p.HSB);

    for (var x = 0; x < p.width; ++x) {
        for (var y = 0; y < p.height; ++y) {
            var signedDistance = sdf(p.createVector(x, y));
            p.stroke(
                Math.abs(signedDistance * 200) % 255,
                signedDistance > 0 ? 100 : 50,
                signedDistance > 0 ? 100 : 50,
            );
            p.line(x, y, x, y); // This is weird but Firefox does not like p.point
        }
    }
};

# %% [markdown]
Now let's define a line segment.

# %%--- [javascript]
# properties:
#   run_on_load: true
# ---%%
var unitSegment = (vec) => {
    var unitVector = createVector(1, 0);
    var width = constrain(vec.x, 0.0, 1.0);
    return -Math.sign(vec.y) * vec.copy().sub(unitVector.mult(width)).mag();
};

var lineSegmentSketch = (p) => {
    var scale = (vec) => {
        return createVector(
            map(vec.x, 0, p.width, -2, 2),
            map(vec.y, 0, p.height, -2, 2),
        );
    };
    p.setup = createSetupFunction(p, (vec) => unitSegment(scale(vec)));
};

p5WithCleanup(lineSegmentSketch, 'lineSegmentSketch');
