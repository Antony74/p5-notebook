# Signed distance function for a star

First we need a way to render an sdf in a p5 sketch

# %%--- [javascript]
# properties:
#   top_hidden: true
#   run_on_load: true
# ---%%
var p5WithCleanup = (sketch, text) => {
    if (!window.sketches) {
        window.sketches = {};
    }

    if (window.sketches[text]) {
        window.sketches[text].remove();
    }

    const matchedElements = Array.from(
        document.querySelectorAll('.cell-bottom'),
    );

    const matchedElement = matchedElements.find(
        (elm) => elm.parentElement.textContent.split(text).length > 1,
    );

    if (!matchedElement) {
        throw new Error(`p5WithCleanup: text not found '${text}'`);
    }

    const p5Instance = new p5(sketch, matchedElement);
    window.sketches[text] = p5Instance;
    return p5Instance;
};

# %%--- [javascript]
# properties:
#   run_on_load: true
# ---%%
await import('/p5-notebook/lib/p5/lib/p5.js');

var notImplemented = () => {
    throw new Error('Not implemented');
};

var createVector = notImplemented;
var map = notImplemented;
var constrain = notImplemented;

var createSetupFunction = (p, sdf) => () => {
    createVector = p.createVector;
    map = p.map;
    constrain = p.constrain;

    p.createCanvas(300, 300);
    p.colorMode(p.HSB);

    for (var x = 0; x < p.width; ++x) {
        for (var y = 0; y < p.height; ++y) {
            var signedDistance = sdf(p.createVector(x, y));
            p.stroke(
                Math.abs(signedDistance * 20) % 255,
                signedDistance > 0 ? 100 : 50,
                signedDistance > 0 ? 100 : 50,
            );
            p.line(x, y, x, y); // This is weird but Firefox does not like p.point
        }
    }
};

# %% [markdown]
Now let's define a line segment.

# %%--- [javascript]
# properties:
#   run_on_load: true
# ---%%
var unitSegment = (vec) => {
    var unitVector = createVector(1, 0);
    var width = constrain(vec.x, 0.0, 1.0);
    var result = vec.copy().sub(unitVector.mult(width)).mag();
    return vec.y >= 0 ? -result : result;
};

# %% [markdown]
Define some simple familiar transformations - scale, translate, and rotate - on an sdf function. Actually we have to apply the inverse transformation to the sdf to get the desired result. These are also known as affine transformations, and can be elegantly and efficient represented by a matrix. I intentionally haven't captured the fully generality here. In particular I want all scale operations have a 1:1 aspect ratio so that scaling the signed distance is trivial.

# %%--- [javascript]
# properties:
#   run_on_load: true
# ---%%
var createScale = (sdf, factor) => (vec) => {
    var signedDistance = sdf(vec.copy().div(factor));
    return signedDistance * factor;
};

var createTranslate = (sdf, translationVec) => (vec) => {
    return sdf(vec.copy().sub(translationVec));
};

var lineSegmentSketch = (p) => {
    var scaleFactor = 100;
    var sdf = createTranslate(
        createScale(unitSegment, scaleFactor),
        p.createVector(100, 150),
    );
    p.setup = createSetupFunction(p, sdf);
};

p5WithCleanup(lineSegmentSketch, 'lineSegmentSketch');

# %% [markdown]
This is an improper sdf, but will become well behaved once we chain line segments to form a closed polygon.
